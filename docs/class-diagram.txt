    %% --- Core Entities & Domain Architecture ---
    User "1" *-- "1" AvailabilitySchedule : Contains
    User "1..*" -- "0..*" Group : Make up a
    User "1" -- "0..1" UserSession : Accesses via
    User "1" *-- "0..*" ExternalCalendarFile : Owns

    AvailabilitySchedule "1" *-- "0..*" TimeSlot : Composed of
    ScheduledMeeting "1" *-- "0..*" TimeSlot : Composed of
    CalendarParser <|.. IcsParser

    User "1" -- "0..*" Invitation : Receives
    User "1" -- "0..*" Vote : Casts
    Group "1" -- "0..*" Invitation : Tied to
    Group "1" -- "0..*" ScheduledMeeting : Has
    Group "1" -- "0..*" Notification : Dispatches
    Group "1" -- "0..*" GroupFileService : manages
    MeetingPoll "1" *-- "0..*" Vote : Aggregates
    MeetingPoll "0..1" -- "1" TimeSlot : Proposes
    MeetingGenerator ..> MeetingPoll
    MeetingGenerator "1" -- "0..*" TimeSlot : Evaluated by
    IcsParser o-- ExternalCalendarFile : Holds reference to

    %% --- Structural Domain Elements ---
    class CalendarParser {
      <<abstract>>
      +validateFormat(fileBytes: byte) bool
      +parseToSlots(fileBytes: byte) List[TimeSlot]
    }

    class TimeSlot {
      +startTime: DateTime
      +endTime: DateTime
      -cadence: str
      +getDuration() long
      +overlaps(other: TimeSlot) bool
      +occursAt(dateTime: DateTime) bool
    }

    %% --- Domain Classes ---
    class User {

      -email: str
      -passwordHash: str
      -firstName: str
      -lastName: str
      -allowEmailReminders: bool
    }

    class AvailabilitySchedule {

    class ScheduledMeeting {

    class MeetingGenerator {

    %% --- Service Classes ---
    class UserService {
      -userRepo: UserRepository
      -profileRepo: UserProfileRepository
      -hasher: PasswordHasher
      +registerUser(email: str, passwordHash: str, firstName: str, lastName: str) User
      +updateProfileData(userId: int, email: str, firstName: str, lastName: str) void
      +deleteUserAccount(userId, password) void
    }

    class AuthenticationService {
      -userRepo: UserRepository
      -sessionManager: SessionManager
      +login(email, password) UserSession
      +logout(token) void
      +validateToken(token) bool
    }

    class PasswordRecoveryService {
      -userRepo: UserRepository
      -tokenProvider: TokenProvider
      -emailService: EmailService
      +requestResetLink(email) void
      +completePasswordReset(token, newPassword) void
    }

    class GroupService {
      -groupRepo: GroupRepository
      -rosterRepo: GroupRosterRepository
      +createGroup(userId, name, desc) Group
      +modifyGroupDetails(groupId, name, desc) void
      +deleteGroup(groupId) void
      +getGroups(userId) list[Group]
    }

    class GroupFileService {
      -fileRepo: GroupFileRepository
      -storageDirectory: str
      +attachFileToGroup(groupId: int, fileName: str, fileBytes: byte) int
      +getFileStream(fileId: int) tuple[str, str]
      +removeFileFromGroup(fileId: int) void
    }

    class GroupMembershipService {

      +leaveGroup(userId: int, groupId: int) void
    }

    class AvailabilityService {

    class MeetingSuggestionService {
